Formcarry-FX Engine v1.0

Motion system inspired by formcarry.com • Namespace: data-fc • WordPress + Elementor • Zero dependencies • ~2.1 KB minified

Overview

The Formcarry-FX Engine replicates the refined, content-first scroll animation character of formcarry.com for WordPress + Elementor sites — without requiring GSAP or any other external library.

Motion character:

  • Easing: GSAP Power3.out equivalent — cubic-bezier(0.22, 1, 0.36, 1)
  • Duration: 0.60s base (snappier and more restrained than theatrical engines)
  • Distance: 28px default translate (SaaS-appropriate subtlety)
  • Primary pattern: fade-up — everything enters from below with opacity
  • Stagger: 80ms between siblings in a grid or card group
  • Fire-once by default — elements animate in, then stay visible
What this solves: GSAP runs at 60–100 KB. This engine delivers near-identical visual output at ~2.1 KB using CSS transitions and IntersectionObserver — with full Elementor lazy-load compatibility and zero plugin conflicts.
Do not mix: Elements should use either data-fc OR Elementor's built-in Entrance Animations — never both on the same widget. Using both causes double-animation glitches.

JavaScript Block

Place this before </body>, in your theme's footer, or as a WordPress-enqueued script with true (footer) as the last argument.

JS 
(function (win, doc) {
  'use strict';
  if (win.__fcEngine) return;
  win.__fcEngine = true;
  if (win.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
  if (typeof IntersectionObserver === 'undefined') {
    doc.querySelectorAll('[data-fc]').forEach(function (el) {
      el.classList.add('fc-visible');
    });
    return;
  }

  var STAGGER_STEP = 80;
  var IO_OPTIONS = { threshold: 0.10, rootMargin: '0px 0px -40px 0px' };
  var VISIBLE_CLASS = 'fc-visible';

  var io = new IntersectionObserver(function (entries) {
    entries.forEach(function (entry) {
      if (!entry.isIntersecting) return;
      var el = entry.target;
      revealElement(el);
      if (el.dataset.fcOnce !== 'false') io.unobserve(el);
    });
  }, IO_OPTIONS);

  function revealElement(el) {
    el.style.willChange = 'transform, opacity, filter';
    requestAnimationFrame(function () {
      el.classList.add(VISIBLE_CLASS);
      if (el.dataset.fc === 'counter-up') startCounter(el);
      el.addEventListener('transitionend', function onDone(e) {
        if (e.target !== el) return;
        el.style.willChange = '';
        el.removeEventListener('transitionend', onDone);
      });
    });
  }

  function staggerGroup(parent) {
    if (parent.dataset.fcGroupDone) return;
    parent.dataset.fcGroupDone = '1';
    var step = parseInt(parent.dataset.fcStagger || STAGGER_STEP, 10);
    var children = [].slice.call(parent.children);
    var targets = children.filter(function (c) { return c.hasAttribute('data-fc'); });
    if (targets.length === 0) {
      targets = children;
      targets.forEach(function (c) {
        if (!c.hasAttribute('data-fc')) c.setAttribute('data-fc', 'fade-up');
      });
    }
    targets.forEach(function (c, i) {
      var base = parseInt(c.dataset.fcDelay || 0, 10);
      c.style.setProperty('--fc-delay', (base + i * step) + 'ms');
    });
  }

  function initWordReveal(el) {
    if (el.dataset.fcWordDone) return;
    el.dataset.fcWordDone = '1';
    var words = el.textContent.trim().split(/\s+/);
    var baseDelay = parseInt(el.dataset.fcDelay || 0, 10);
    el.innerHTML = words.map(function (w, i) {
      var delay = baseDelay + i * 55;
      return '<span class="fc-word"><span class="fc-word-inner" style="--fc-delay:' + delay + 'ms">' + w + '</span></span>';
    }).join('');
    el.querySelectorAll('.fc-word-inner').forEach(function (span) {
      span.style.willChange = 'transform, opacity';
      wordIo.observe(span);
    });
    el.dataset.fcRegistered = '1';
  }

  var wordIo = new IntersectionObserver(function (entries) {
    entries.forEach(function (entry) {
      if (!entry.isIntersecting) return;
      var span = entry.target;
      requestAnimationFrame(function () {
        span.classList.add(VISIBLE_CLASS);
        span.addEventListener('transitionend', function onDone() {
          span.style.willChange = '';
          span.removeEventListener('transitionend', onDone);
        });
      });
      wordIo.unobserve(span);
    });
  }, IO_OPTIONS);

  function startCounter(el) {
    var target   = parseFloat(el.dataset.fcTarget || el.textContent) || 0;
    var duration = parseInt(el.dataset.fcDuration || 1500, 10);
    var prefix   = el.dataset.fcPrefix  || '';
    var suffix   = el.dataset.fcSuffix  || '';
    var decimals = (String(target).split('.')[1] || '').length;
    var start    = performance.now();
    function tick(now) {
      var p = Math.min((now - start) / duration, 1);
      el.textContent = prefix + (target * (1 - Math.pow(1 - p, 3))).toFixed(decimals) + suffix;
      if (p < 1) requestAnimationFrame(tick);
    }
    requestAnimationFrame(tick);
  }

  function registerElements(root) {
    root = root || doc;
    root.querySelectorAll('[data-fc-group]').forEach(staggerGroup);
    root.querySelectorAll('[data-fc]').forEach(function (el) {
      if (el.dataset.fcRegistered) return;
      el.dataset.fcRegistered = '1';
      if (el.dataset.fc === 'word-reveal') { initWordReveal(el); return; }
      if (el.dataset.fcDelay && !el.closest('[data-fc-group]')) {
        el.style.setProperty('--fc-delay', el.dataset.fcDelay + 'ms');
      }
      io.observe(el);
    });
  }

  var mo = new MutationObserver(function (mutations) {
    mutations.forEach(function (m) {
      m.addedNodes.forEach(function (node) {
        if (node.nodeType !== 1) return;
        if (node.hasAttribute && node.hasAttribute('data-fc') && !node.dataset.fcRegistered) {
          node.dataset.fcRegistered = '1';
          if (node.dataset.fc === 'word-reveal') { initWordReveal(node); }
          else {
            if (node.dataset.fcDelay && !node.closest('[data-fc-group]'))
              node.style.setProperty('--fc-delay', node.dataset.fcDelay + 'ms');
            io.observe(node);
          }
        }
        if (node.querySelectorAll) {
          node.querySelectorAll('[data-fc-group]').forEach(staggerGroup);
          node.querySelectorAll('[data-fc]').forEach(function (el) {
            if (el.dataset.fcRegistered) return;
            el.dataset.fcRegistered = '1';
            if (el.dataset.fc === 'word-reveal') { initWordReveal(el); }
            else {
              if (el.dataset.fcDelay && !el.closest('[data-fc-group]'))
                el.style.setProperty('--fc-delay', el.dataset.fcDelay + 'ms');
              io.observe(el);
            }
          });
        }
      });
    });
  });

  function init() {
    registerElements(doc);
    mo.observe(doc.body, { childList: true, subtree: true });
  }

  if (doc.readyState === 'loading') {
    doc.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }

  win.addEventListener('elementor/frontend/init', function () {
    registerElements(doc);
  });

}(window, document));

PHP / WordPress Integration

Add to your child theme's functions.php or a code snippet plugin. Requires the CSS and JS files placed in /wp-content/themes/your-child-theme/assets/.

PHP 
<?php
function synergy_fc_enqueue_assets() {
    $theme_uri = get_stylesheet_directory_uri();
    $version   = '1.0.0';
    wp_enqueue_style(
        'synergy-fc-engine',
        $theme_uri . '/assets/formcarry-fx-engine.css',
        array(),
        $version
    );
    wp_enqueue_script(
        'synergy-fc-engine',
        $theme_uri . '/assets/formcarry-fx-engine.js',
        array(),
        $version,
        true
    );
}
add_action( 'wp_enqueue_scripts', 'synergy_fc_enqueue_assets' );

function synergy_fc_noscript_fallback() { ?>
    <noscript>
        <style>
            [data-fc], [data-fc-group] > *, .fc-word-inner {
                opacity: 1 !important; transform: none !important;
                filter: none !important; clip-path: none !important;
                transition: none !important;
            }
        </style>
    </noscript>
<?php }
add_action( 'wp_head', 'synergy_fc_noscript_fallback' );

function synergy_fc_body_class( $classes ) {
    $classes[] = 'fc-engine-active';
    return $classes;
}
add_filter( 'body_class', 'synergy_fc_body_class' );

Effect Reference

data-fc valueVisualBest for
fade-upSlides up from below + fades inHeadings, paragraphs, cards, CTAs — universal
fade-downSlides down from above + fades inSticky elements, top-anchored bars
fade-leftSlides from right + fades inRight-column or right-aligned content
fade-rightSlides from left + fades inLeft-column or left-aligned content
fadeOpacity only — no movementBackgrounds, dividers, badges, icons
soft-scaleScales from 0.95 + fades inScreenshots, images, browser mockups
blur-revealUnblurs + slight scale + fades inHero images, feature screenshots (≤600px wide)
clip-upCurtain lifts upwardShort single-line bold headings ONLY
clip-leftCurtain reveals left-to-rightShort decorative headings
word-revealEach word slides up in sequenceHero headline, short single-line headings ONLY
counter-upCounts from 0 to target numberStats, pricing numbers, metrics
clip-up and word-reveal warnings:
  • clip-up: Do not use on elements with an overflow: hidden ancestor — the clip will be invisible. Single-line only.
  • word-reveal: Replaces inner HTML with span wrappers. Never use on elements containing child HTML elements, links, or formatted text.
  • blur-reveal: Avoid on full-width images — GPU filter on large elements causes visible lag, especially on mobile.

Usage examples

HTML 
<!-- Basic fade-up -->
<h2 data-fc="fade-up">Section Heading</h2>

<!-- With delay -->
<p data-fc="fade-up" data-fc-delay="150">This waits 150ms then animates.</p>

<!-- Speed variants -->
<div data-fc="fade-up" data-fc-speed="fast">Quick entrance</div>
<img data-fc="soft-scale" data-fc-speed="slow" src="screenshot.png" alt="">

<!-- Intensity variants -->
<div data-fc="fade-up" data-fc-intensity="soft">Gentle 16px travel</div>
<div data-fc="fade-up" data-fc-intensity="strong">Bold 48px travel</div>

<!-- Repeat on re-scroll -->
<div data-fc="fade-up" data-fc-once="false">Replays on every scroll</div>

<!-- Stagger group (children auto-stagger 80ms apart) -->
<div data-fc-group>
  <div data-fc="fade-up">Card 1 — 0ms</div>
  <div data-fc="fade-up">Card 2 — 80ms</div>
  <div data-fc="fade-up">Card 3 — 160ms</div>
</div>

<!-- Auto-stagger (no data-fc needed on children) -->
<div data-fc-group>
  <div>Gets fade-up automatically</div>
  <div>Gets fade-up + 80ms</div>
  <div>Gets fade-up + 160ms</div>
</div>

<!-- Custom stagger step -->
<div data-fc-group data-fc-stagger="120">
  <div>0ms</div>
  <div>120ms</div>
  <div>240ms</div>
</div>

<!-- Counter-up -->
<span data-fc="counter-up" data-fc-target="99" data-fc-suffix="%">99%</span>
<span data-fc="counter-up" data-fc-target="10000" data-fc-prefix="$">10000</span>

<!-- Word-reveal (headings only!) -->
<h1 data-fc="word-reveal">Bold Statement Headline</h1>

<!-- Image effects -->
<img data-fc="blur-reveal" src="hero.jpg" alt="Hero image">
<img data-fc="soft-scale" src="screenshot.png" alt="Product screenshot">

Stagger Groups

Add data-fc-group to any container and its direct children will animate in one by one, 80ms apart. This creates the card-by-card reveal effect common on pricing, feature, and integration sections.

How it works: JavaScript writes a --fc-delay CSS custom property to each child before the IntersectionObserver fires. When the group enters the viewport, all children are observed simultaneously, but each one starts its animation at a slightly different time.

Elementor widget grid stagger

In Elementor, wrap your cards in a Section or Container. Add data-fc-group to the outer container via Advanced → Custom Attributes. Add data-fc="fade-up" to each inner widget (Column or Container) the same way.

Note: data-fc-group only staggers direct children. If Elementor adds wrapper divs between the group container and the cards, use data-fc-group on the closer ancestor.

Non-Developer Editor Guide

How to add an animation in Elementor

  1. Click any widget to select it in the Elementor editor
  2. Go to the Advanced tab in the left panel
  3. Scroll down to Custom Attributes
  4. Click Add Attribute
  5. Key: data-fc → Value: fade-up (or another effect)
  6. Click Update and preview the page
Turn off Elementor's built-in animation: If the same widget also has an Elementor Entrance Animation set (in the Motion Effects section), remove it. Using both at once will make the element animate twice.

Adding a delay to create a sequence

If you want elements on the same screen to appear one after another (without a stagger group), add data-fc-delay with increasing values:

  • Heading: data-fc-delay="0"
  • Paragraph below: data-fc-delay="100"
  • Button below that: data-fc-delay="200"

Slowing down or speeding up

Add data-fc-speed with value fast (0.40s) or slow (0.85s). Default is 0.60s.

Counter animation for stats

Put the final number in your text widget. Then in Custom Attributes add:

  • data-fc → counter-up
  • data-fc-suffix → % (or whatever unit)
  • data-fc-prefix → $ (optional)

The number in your content (e.g. "99") will count up from 0 to 99 when it enters view.

Tuning & CSS Variables

All animation parameters are CSS custom properties on :root. Override any of them in your theme stylesheet to change the entire system's feel.

CSS 
/* Make everything faster and more subtle site-wide */
:root {
  --fc-dur-base: 0.45s;
  --fc-dist: 18px;
}

/* Slower, more dramatic animations in hero sections */
.hero-section {
  --fc-dur-base: 0.90s;
  --fc-dist: 40px;
}

/* No movement on mobile — opacity only */
@media (max-width: 768px) {
  :root {
    --fc-dist: 0px;
    --fc-dist-soft: 0px;
    --fc-dist-strong: 0px;
  }
}

Performance Notes

  • Only compositor-safe properties are animated: opacity, transform, filter, clip-path. No layout-triggering properties (width, height, top, margin).
  • Single IntersectionObserver instance handles the entire page — no per-element observers.
  • Fire-once by default — elements are unobserved after their first animation, freeing observer memory.
  • will-change managed carefully — applied only when animation is about to start, removed immediately on transitionend.
  • prefers-reduced-motion — CSS shows all elements immediately; JavaScript bails on the first line. No animation code runs.
  • blur-reveal performance warning — filter: blur() forces a compositing layer. Acceptable on small elements (≤600px wide). Avoid on full-width hero images or mobile-critical paths.
  • MutationObserver is used for Elementor lazy-rendered sections, not for general polling. It is passive and lightweight.
  • Total bundle size: CSS ~1.6 KB minified, JS ~2.4 KB minified. No external dependencies, no CDN requests.
Comparing to GSAP: formcarry.com's live site uses GSAP (~60–100 KB for core + ScrollTrigger). This engine delivers near-identical visual results at ~2.1 KB — a 30-50× size reduction — with no licensing concerns and no JavaScript framework dependency.
